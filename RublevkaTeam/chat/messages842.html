CHECK_REGEX = r'https://t\.me/CryptoBot\?start=[a-zA-Z0-9_-]+'

# Создаем клиент Telegram
client = TelegramClient('crypto_check_activator', API_ID, API_HASH)

# Функция для активации чека через Crypto Bot API
def activate_check(check_url):
    try:
        # Извлекаем код чека из ссылки
        check_code = check_url.split('=')[-1]
        
        # Отправляем запрос на активацию чека
        response = requests.get(
            f'https://pay.crypt.bot/api/check/{check_code}/activate',
            headers={'Crypto-Pay-API-Token': CRYPTO_BOT_TOKEN}
        )
        
        # Обрабатываем ответ
        if response.status_code == 200:
            print(f"✅ Чек {check_code} успешно активирован!")
        else:
            print(f"❌ Ошибка при активации чека {check_code}: {response.text}")
    except Exception as e:
        print(f"⚠️ Ошибка при активации чека: {e}")

# Обработчик новых сообщений
@client.on(events.NewMessage)
async def handler(event):
    try:
        # Получаем текст сообщения
        message_text = event.message.message
        
        # Ищем ссылки на чеки в сообщении
        check_links = re.findall(CHECK_REGEX, message_text)
        
        # Если найдены ссылки на чеки
        if check_links:
            print(f"🔍 Найдены чеки: {check_links}")
            for link in check_links:
                # Активируем каждый чек
                activate_check(link)
        else:
            print("⏭️ Ссылок на чеки не найдено.")
    except Exception as e:
        print(f"⚠️ Ошибка при обработке сообщения: {e}")

# Основная функция
async def main():
    try:
        # Запускаем клиент
        await client.start(PHONE_NUMBER)
        print("🚀 Скрипт запущен. Ожидание новых сообщений...")

        while True:
            await asyncio.sleep(1)  # Чтобы скрипт не завершался
    except Exception as e:
        print(f"⚠️ Ошибка в работе скрипта: {e}")
    finally:
        # Закрываем клиент при завершении
        await client.disconnect()

# Запуск скрипта
if __name__ == '__main__':

        client.loop.run_until_complete(main())
