        function showMoreInfo(userId) {
            // Fetch user data using AJAX
            fetch(`/worker/users/${userId}/info`)
                .then(response => response.json())
                .then(data => {
                    let countryInfo = 'N/A';

                    if (data.country && typeof data.country === 'object') {
                        const flag = data.country.flag ? `<img src="${data.country.flag}"
                    alt="${data.country.name || 'Unknown'}"
                    title="${data.country.name || 'Unknown'}"
                    style="width: 30px; height: 15px; object-fit: cover; margin-right: 5px;">` : '';

                        countryInfo = `${flag} ${data.country.name || 'Unknown'}`;
                    } else if (data.country_id) {
                        countryInfo = data.country_id;
                    }

                    const userInfo = `
                <strong>Username:</strong> ${data.username}<br>
                <strong>Email:</strong> ${data.email}<br>
                <strong>Created:</strong> ${data.created_at ?? 'N/A'}<br>
                <strong>Domain:</strong> ${data.domain}<br>
                <strong>Country:</strong> ${countryInfo}<br>
                <strong>Last Online:</strong> ${data.last_online ?? 'Never'}<br>
                <strong>Balance:</strong> ${data.balance}<br>
                <strong>Remaining Balance:</strong> ${data.remaining_balance}<br>
                <strong>Wins Count:</strong> ${data.wins_count}<br>
                <strong>Bet Count:</strong> ${data.bet_count}<br>
                <strong>Win Streak:</strong> ${data.win_streak}<br>
                <strong>Lose Streak:</strong> ${data.lose_streak}<br>
                <strong>Last Game Type:</strong> ${data.last_game_type}<br>
                <strong>Total Profit:</strong> ${data.total_profit}<br>

                <!-- Ban User Toggle -->
                <div class="form-check form-switch mt-3">
                    <input class="form-check-input" type="checkbox" id="banUserToggle"
                        ${data.is_banned ? 'checked' : ''}
                        onchange="toggleBan(${data.id}, this.checked)">
                    <label class="form-check-label" for="banUserToggle"><strong>Ban User</strong></label>
                </div>

                <!-- Ban by IP & Cookies Toggle -->
                <div class="form-check form-switch mt-2">
                    <input class="form-check-input" type="checkbox" id="banIPToggle"
                        ${data.is_banned_by_ip ? 'checked' : ''}
                        onchange="toggleBanIP(${data.id}, this.checked)">
                    <label class="form-check-label" for="banIPToggle"><strong>Ban by IP & Cookies</strong></label>
                </div>
            `;


                    document.getElementById('userInfo').innerHTML = userInfo;
                    // Show the modal
                    $('#userModal').modal('show');
                })
                .catch(error => {
                    console.error('Error fetching user info:', error);
                });
        }
        function toggleBan(userId, isBanned) {
            fetch(`/worker/users/${userId}/ban`, {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "X-CSRF-TOKEN": document.querySelector('meta[name="csrf-token"]').getAttribute("content"),
                },
                body: JSON.stringify({ ban: isBanned }),
            })
                .then(response => response.json())
                .then(data => {
                    if (data.success) {
                        alert(isBanned ? "User banned successfully." : "User unbanned.");
                    } else {
                        alert("Error banning user.");
                    }
                })
                .catch(error => console.error("Error:", error));
        }

        function toggleBanIP(userId, isBannedByIP) {
            fetch(`/worker/users/${userId}/ban-ip`, {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "X-CSRF-TOKEN": document.querySelector('meta[name="csrf-token"]').getAttribute("content"),
                },
                body: JSON.stringify({ ban_ip: isBannedByIP }),
            })
                .then(response => response.json())
                .then(data => {
                    if (data.success) {
                        alert(isBannedByIP ? "User banned by IP & Cookies." : "User unbanned from IP & Cookies.");
                    } else {
                        alert("Error banning user by IP.");
                    }
                })
                .catch(error => console.error("Error:", error));
        }
